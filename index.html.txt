<script>
    // 转盘选项：8个分区
    const items = [
        {type:"真心话",text:"说出你最后一次撒谎的事情"},
        {type:"大冒险",text:"对着窗外大喊三声我最帅/最美"},
        {type:"真心话",text:"在场最喜欢哪一个人"},
        {type:"大冒险",text:"做5个搞怪表情包拍照"},
        {type:"真心话",text:"偷偷吐槽一个在场的人"},
        {type:"大冒险",text:"模仿大猩猩走路一圈"},
        {type:"真心话",text:"谈过几次恋爱"},
        {type:"大冒险",text:"喝一口饮料然后憋气10秒"}
    ];
    const colors = ['#ff6b6b','#4ecdc4','#45b7d1','#ffe066','#ff8fab','#95e1d3','#f38181','#aa96da'];

    const canvas = document.getElementById('wheel');
    const ctx = canvas.getContext('2d');
    const btn = document.getElementById('startBtn');
    const mask = document.getElementById('mask');
    const resultText = document.getElementById('resultText');

    let isRunning = false;
    let rotateDeg = 0;
    const singleAngle = 360 / items.length;

    // 绘制转盘
    function drawWheel(){
        ctx.clearRect(0,0,400,400);
        for(let i=0;i<items.length;i++){
            const start = i * singleAngle * Math.PI / 180;
            const end = (i+1) * singleAngle * Math.PI / 180;
            ctx.beginPath();
            ctx.moveTo(200,200);
            ctx.arc(200,200,200,start,end);
            ctx.closePath();
            ctx.fillStyle = colors[i];
            ctx.fill();
            ctx.strokeStyle="#fff";
            ctx.lineWidth=2;
            ctx.stroke();

            // 文字
            ctx.save();
            ctx.translate(200,200);
            ctx.rotate(start + singleAngle/2);
            ctx.font="bold 14px 微软雅黑";
            ctx.fillStyle="#fff";
            ctx.textAlign="right";
            ctx.fillText(items[i].type,170,6);
            ctx.restore();
        }
    }
    drawWheel();

    // 点击旋转
    btn.onclick = function(){
        if(isRunning) return;
        isRunning = true;
        btn.disabled = true;
        // 随机旋转圈数
        const addDeg = 360 * (5+Math.floor(Math.random()*6)) + Math.floor(Math.random()*360);
        rotateDeg += addDeg;
        canvas.style.transform = `rotate(${rotateDeg}deg)`;

        // 4秒后停止计算结果
        setTimeout(()=>{
            isRunning = false;
            btn.disabled = false;
            // 计算最终停留分区
            const finalDeg = rotateDeg % 360;
            const index = Math.floor((360 - finalDeg) / singleAngle) % items.length;
            const res = items[index];
            resultText.innerText = `${res.type}：${res.text}`;
            mask.style.display = "flex";
        },4000);
    }

    function closeModal(){
        mask.style.display = "none";
    }
</script>
